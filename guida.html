🔧 1. Setup XAMPP e Configurazione

Cos'è XAMPP?
XAMPP è un pacchetto che include Apache (server web), MySQL (database), PHP e Perl. Ti permette di creare un ambiente di sviluppo locale sul tuo computer.

Creazione Database e Tabella

-- Crea il database CREATE DATABASE mio_sito; -- Seleziona il database USE mio_sito; -- Crea tabella utenti CREATE TABLE utenti ( id INT AUTO_INCREMENT PRIMARY KEY, nome VARCHAR(50) NOT NULL, email VARCHAR(100) UNIQUE NOT NULL, eta INT, data_registrazione TIMESTAMP DEFAULT CURRENT_TIMESTAMP ); -- Inserisci dati di esempio INSERT INTO utenti (nome, email, eta) VALUES ('Mario Rossi', 'mario@email.com', 30), ('Laura Bianchi', 'laura@email.com', 25), ('Giuseppe Verdi', 'giuseppe@email.com', 35);

🔗 3. Connessione PHP al Database

File di Configurazione (config.php)

<?php // config.php - File di configurazione database // Parametri di connessione define('DB_HOST', 'localhost'); define('DB_NAME', 'mio_sito'); define('DB_USER', 'root'); define('DB_PASS', ''); // XAMPP di default ha password vuota // Funzione per connessione con PDO (raccomandato) function connetti_database() { try { $dsn = "mysql:host=" . DB_HOST . ";dbname=" . DB_NAME . ";charset=utf8"; $pdo = new PDO($dsn, DB_USER, DB_PASS); // Configura PDO per lanciare eccezioni $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION); return $pdo; } catch(PDOException $e) { die("Errore connessione: " . $e->getMessage()); } } // Connessione alternativa con MySQLi function connetti_mysqli() { $conn = new mysqli(DB_HOST, DB_USER, DB_PASS, DB_NAME); if ($conn->connect_error) { die("Connessione fallita: " . $conn->connect_error); } return $conn; } ?>
✅ PDO vs MySQLi:
- PDO: Più moderno, supporta più database, sintassi più pulita
- MySQLi: Più veloce con MySQL, interfaccia procedurale e orientata agli oggetti

File PHP per estrarre dati (database.php)

<?php // database.php - File per estrarre dati dal database require_once 'config.php'; // Funzione per ottenere tutti gli utenti function ottieni_utenti() { $pdo = connetti_database(); try { $stmt = $pdo->prepare("SELECT * FROM utenti ORDER BY nome"); $stmt->execute(); $utenti = $stmt->fetchAll(PDO::FETCH_ASSOC); return $utenti; } catch(PDOException $e) { error_log("Errore query: " . $e->getMessage()); return []; } } // Funzione per cercare utente per ID function ottieni_utente_per_id($id) { $pdo = connetti_database(); try { $stmt = $pdo->prepare("SELECT * FROM utenti WHERE id = ?"); $stmt->execute([$id]); $utente = $stmt->fetch(PDO::FETCH_ASSOC); return $utente; } catch(PDOException $e) { error_log("Errore query: " . $e->getMessage()); return null; } } // Se il file viene chiamato direttamente, restituisci JSON if (basename($_SERVER['PHP_SELF']) == 'database.php') { header('Content-Type: application/json'); if (isset($_GET['azione'])) { switch ($_GET['azione']) { case 'tutti_utenti': echo json_encode(ottieni_utenti()); break; case 'utente': $id = $_GET['id'] ?? 0; echo json_encode(ottieni_utente_per_id($id)); break; default: echo json_encode(['errore' => 'Azione non valida']); } } else { echo json_encode(ottieni_utenti()); } } ?>

🔍 5. Query SQL Avanzate

Tipi di Query Essenziali

Operazione SQL Uso
SELECT SELECT * FROM tabella Leggere dati
INSERT INSERT INTO tabella VALUES(...) Inserire dati
UPDATE UPDATE tabella SET campo=valore Aggiornare dati
DELETE DELETE FROM tabella WHERE... Cancellare dati

Esempi di Query Avanzate

-- Ricerca con filtri SELECT * FROM utenti WHERE eta BETWEEN 25 AND 35 AND nome LIKE 'M%' ORDER BY data_registrazione DESC; -- Contare records SELECT COUNT(*) as totale_utenti FROM utenti; -- Raggruppamento SELECT eta, COUNT(*) as numero_utenti FROM utenti GROUP BY eta HAVING COUNT(*) > 1; -- Join tra tabelle (se hai più tabelle) SELECT u.nome, u.email, p.titolo as ultimo_post FROM utenti u LEFT JOIN posts p ON u.id = p.user_id WHERE p.data_creazione = ( SELECT MAX(data_creazione) FROM posts WHERE user_id = u.id ); -- Ricerca full-text SELECT * FROM utenti WHERE MATCH(nome, email) AGAINST('mario' IN NATURAL LANGUAGE MODE);

File PHP con Query Avanzate (query_avanzate.php)

<?php require_once 'config.php'; class UtentiManager { private $pdo; public function __construct() { $this->pdo = connetti_database(); } // Ricerca utenti con filtri public function ricerca_utenti($filtri = []) { $sql = "SELECT * FROM utenti WHERE 1=1"; $params = []; if (!empty($filtri['nome'])) { $sql .= " AND nome LIKE ?"; $params[] = '%' . $filtri['nome'] . '%'; } if (!empty($filtri['eta_min'])) { $sql .= " AND eta >= ?"; $params[] = $filtri['eta_min']; } if (!empty($filtri['eta_max'])) { $sql .= " AND eta <= ?"; $params[] = $filtri['eta_max']; } $sql .= " ORDER BY nome"; $stmt = $this->pdo->prepare($sql); $stmt->execute($params); return $stmt->fetchAll(PDO::FETCH_ASSOC); } // Statistiche utenti public function ottieni_statistiche() { $stats = []; // Totale utenti $stmt = $this->pdo->query("SELECT COUNT(*) as totale FROM utenti"); $stats['totale'] = $stmt->fetch()['totale']; // Età media $stmt = $this->pdo->query("SELECT AVG(eta) as eta_media FROM utenti"); $stats['eta_media'] = round($stmt->fetch()['eta_media'], 1); // Utenti per fascia d'età $stmt = $this->pdo->query(" SELECT CASE WHEN eta < 25 THEN 'Under 25' WHEN eta BETWEEN 25 AND 35 THEN '25-35' ELSE 'Over 35' END as fascia, COUNT(*) as numero FROM utenti GROUP BY fascia "); $stats['fasce_eta'] = $stmt->fetchAll(PDO::FETCH_ASSOC); return $stats; } // Inserimento nuovo utente public function inserisci_utente($dati) { $sql = "INSERT INTO utenti (nome, email, eta) VALUES (?, ?, ?)"; $stmt = $this->pdo->prepare($sql); try { $stmt->execute([ $dati['nome'], $dati['email'], $dati['eta'] ]); return $this->pdo->lastInsertId(); } catch(PDOException $e) { if ($e->getCode() == 23000) { // Violazione vincolo UNIQUE throw new Exception("Email già esistente"); } throw $e; } } } // API endpoints if ($_SERVER['REQUEST_METHOD'] === 'GET') { $manager = new UtentiManager(); header('Content-Type: application/json'); switch ($_GET['azione'] ?? '') { case 'ricerca': $filtri = [ 'nome' => $_GET['nome'] ?? '', 'eta_min' => $_GET['eta_min'] ?? '', 'eta_max' => $_GET['eta_max'] ?? '' ]; echo json_encode($manager->ricerca_utenti($filtri)); break; case 'statistiche': echo json_encode($manager->ottieni_statistiche()); break; default: echo json_encode(['errore' => 'Azione non valida']); } } if ($_SERVER['REQUEST_METHOD'] === 'POST') { $manager = new UtentiManager(); header('Content-Type: application/json'); $input = json_decode(file_get_contents('php://input'), true); try { $id = $manager->inserisci_utente($input); echo json_encode(['successo' => true, 'id' => $id]); } catch(Exception $e) { echo json_encode(['errore' => $e->getMessage()]); } } ?>
